function showPage(id, btn) {
  document.querySelectorAll('.page').forEach(p => p.classList.remove('show'));
  document.getElementById(id).classList.add('show');
  document.querySelectorAll('nav button').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
}

// --- Symptom checker ---
function analyze() {
  const sel = Array.from(document.getElementById('symptom').selectedOptions).map(o => o.value);
  const box = document.getElementById('diag-result');
  box.style.display = 'block';
  if (!sel.length) { box.className='result warn'; box.textContent='Please select at least one symptom.'; return; }
  let f = [];
  if (sel.includes('fatigue') && sel.includes('pica')) f.push('Possible Iron Deficiency Anemia.');
  if (sel.includes('dry_skin')) f.push('Possible Vitamin A or E deficiency.');
  if (sel.includes('mouth')) f.push('Possible Vitamin C or B-vitamin deficiency.');
  if (sel.includes('growth')) f.push('General malnutrition risk — consult a pediatrician.');
  if (!f.length) f.push('No specific pattern found in this demo. Consult a doctor.');
  box.className = 'result ok';
  box.innerHTML = '<strong>Preliminary Analysis:</strong><br>' + f.join('<br>');
}

// --- BMI ---
function calcBMI() {
  const h = document.getElementById('height').value, w = document.getElementById('weight').value;
  const box = document.getElementById('bmi-result');
  box.style.display = 'block';
  if (!h || !w) { box.className='result warn'; box.textContent='Enter both height and weight.'; return; }
  const bmi = (w / ((h/100)**2)).toFixed(1);
  let status, cls;
  if (bmi < 15) { status='Underweight'; cls='warn'; }
  else if (bmi < 25) { status='Healthy Weight Range'; cls='ok'; }
  else { status='Overweight Risk'; cls='bad'; }
  box.className = 'result ' + cls;
  box.innerHTML = `BMI: <strong>${bmi}</strong> — ${status}<br><small>For children, age/gender percentiles are needed for accuracy.</small>`;
}
